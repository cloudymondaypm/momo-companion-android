cmake_minimum_required(VERSION 3.16)
project(fold5_audio C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory(opus-1.5.2)
add_library(fold5_opus SHARED opus_jni.c)
target_include_directories(fold5_opus PRIVATE opus-1.5.2/include)
target_link_libraries(fold5_opus PRIVATE opus m)
target_link_options(fold5_opus PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
